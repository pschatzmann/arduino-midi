107 virtual bool tick(uint32_t timestamp) {
108 bool active = false;
109 // flush output buffers and synchronize clocks
110 applemidi_tick();
111 int packetSize = udpControl.parsePacket();
112 remote_port = udpControl.remotePort();
113 if (packetSize > 0) {
114 uint8_t ip[4];
115 toBytes(udpControl.remoteIP(), ip);
116 int len = udpControl.read(rx_buffer, MIDI_BUFFER_SIZE);
117 MIDI_LOGD("control: %d -> %d ", remote_port, len);
118 if (len > 0)
119 applemidi_parse_udp_datagram(ip, remote_port, rx_buffer, len, false);
120 active = true;
121 }
122 packetSize = udpData.parsePacket();
123 if (packetSize > 0) {
124 int data_remote_port = udpData.remotePort();
125 uint8_t ip[4];
126 toBytes(udpData.remoteIP(), ip);
127 int len = udpData.read(rx_buffer, MIDI_BUFFER_SIZE);
128 MIDI_LOGD("data: %d -> %d", data_remote_port, len);
129 if (len > 0)
130 applemidi_parse_udp_datagram(ip, data_remote_port, rx_buffer, len,
131 true);
132 active = true;
133 }
134 return active;
135 }

202 static int32_t applemidi_if_send_udp_datagram(uint8_t* ip_addr, uint16_t port,
203 uint8_t* tx_data,
204 size_t tx_len) {
205 MIDI_LOGD("applemidi_if_send_udp_datagram: port=%d", port);
206 if (_instance) {
207 IPAddress address(ip_addr[0], ip_addr[1], ip_addr[2], ip_addr[3]);
208 UDPClass* p_udp = port == _instance->remote_port
209 ? &(_instance->udpControl)
210 : &(_instance->udpData);
211 p_udp->beginPacket(address, port);
212 int32_t result = p_udp->write(tx_data, tx_len);
213 p_udp->endPacket();
214 return result;
215 }
216 return 0;
217 }
